{{/* Discover coin-face images from static/img/coin-faces/. Only images that have a .webp companion are emitted: coin-flip.js builds an image-set() with a .webp source at runtime, so a source without one would hand webp-capable browsers a 404 and render a blank coin face. scripts/validate-coin-faces.sh enforces the companion so this never silently drops a face the author meant to include. */}} {{ $coinDir := "static/img/coin-faces" }} {{ $coinFaces := slice }} {{ if fileExists $coinDir }} {{ range readDir $coinDir }} {{ if not .IsDir }} {{ $lower := .Name | lower }} {{ if or (strings.HasSuffix $lower ".png") (strings.HasSuffix $lower ".jpg") (strings.HasSuffix $lower ".jpeg") }} {{/* Derive the companion from .Name, not the lowercased copy: on a case-sensitive filesystem Me.PNG needs Me.webp, and checking me.webp would skip the face while the validator passed it. */}} {{ $webpName := .Name | replaceRE `(?i)\.(png|jpg|jpeg)$` ".webp" }} {{ if fileExists (printf "%s/%s" $coinDir $webpName) }} {{ $coinFaces = $coinFaces | append (printf "/img/coin-faces/%s" .Name) }} {{ else }} {{ warnf "coin-faces: skipping %q — no .webp companion. Generate one: cwebp -q 80 -m 6 %s/%s -o %s/%s" .Name $coinDir .Name $coinDir $webpName }} {{ end }} {{ end }} {{ end }} {{ end }} {{ else }} {{ warnf "coin-faces: %s does not exist — the coin will show only the front face" $coinDir }} {{ end }} {{/* Both faces come from the same pool — a photograph and an illustration have equal standing here, and coin-flip.js re-randomises both on load. These two are only what the first paint shows before the script runs. */}} {{ $faceA := "" }}{{ $faceB := "" }} {{ if gt (len $coinFaces) 0 }}{{ $faceA = index $coinFaces 0 }}{{ end }} {{ if gt (len $coinFaces) 1 }}{{ $faceB = index $coinFaces 1 }}{{ end }}
